# TrackerMultimedia
#
# La regla de reescritura de SPA (`public/_redirects`) hace que cualquier ruta
# devuelva 200 con el HTML de la aplicación, también las privadas. Para un
# rastreador eso significa que /library o /profile "existen" y son indexables,
# aunque su contenido real solo llegue tras autenticarse. De ahí este archivo.
#
# Nota de alcance: robots.txt pide, no impide. No es un control de acceso —eso lo
# hace el backend, que exige un token en cada endpoint—; sirve para que las rutas
# privadas no acaben listadas en un buscador.

User-agent: *

# Zona privada: nada que un buscador deba listar.
Disallow: /library
Disallow: /catalog
Disallow: /profile

# Pantallas que solo tienen sentido con un token en la URL. Indexarlas no aporta
# nada y expone en el buscador enlaces que caducan.
Disallow: /reset-password
Disallow: /confirm-email
Disallow: /oauth-callback
Disallow: /link-account
Disallow: /resend-confirmation

# Públicas y con contenido propio: /login y /register quedan permitidas.

Sitemap: https://trackermultimedia.netlify.app/sitemap.xml
